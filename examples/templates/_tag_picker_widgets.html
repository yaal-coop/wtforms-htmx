{# TagPickerForm.tags: chips for selected, catalog for unselected. Catalog buttons post the tag to ``add_tag`` (value_add); chip × buttons post it to ``rm_tag`` (value_remove). Persisted across submits via multi-valued hidden inputs. #} {%- macro tags(field) -%} {%- set selected = field.data or [] -%} {%- set choices = field.iter_choices() -%}
{% for choice in choices if choice.value not in selected %} {{ field._form.add_tag(value=choice.value, label=choice.label, id=False, class_="tag tag-button") }} {% endfor %} {% if selected|length == choices|length %} All tags selected. {% endif %}
{% for value in selected %} {{ value }} {{ field._form.rm_tag(value=value, label="×", id=False, class_="chip-remove", aria_label="Remove " ~ value) }} {% else %} No tags selected yet. {% endfor %}
{%- endmacro %}
